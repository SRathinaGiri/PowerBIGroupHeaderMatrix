 150     public update(options: VisualUpdateOptions) {
 151         const dataView: DataView | undefined = options.dataViews && options.dataViews[0];
 152         this.formattingSettings = this.formattingSettingsService.populateFormattingSettingsModel(VisualFormattingSettingsModel, dataView);
 153         // Apply sticky preference if present
 154         const sticky = this.getObjectValue<boolean>(dataView?.metadata?.objects, "state", "stickyHeaders", true);
 155         this.container.classList.toggle("ghm-sticky", !!sticky);
 156         this.repeatLabels = this.getObjectValue<boolean>(dataView?.metadata?.objects, "state", "repeatLabels", false);
 157         this.rowHeaderMinWidth = this.getObjectValue<number>(dataView?.metadata?.objects, "state", "rowHeaderMinWidth", 160) || 160;
 158         this.rowHeaderFontSize = this.getObjectValue<number>(dataView?.metadata?.objects, "labels", "rowHeaderFontSize", 12) || 12;
 159         this.rowHeaderFontFamily = this.getObjectValue<string>(dataView?.metadata?.objects, "labels", "rowHeaderFontFamily", "");
 160         this.colHeaderFontSize = this.getObjectValue<number>(dataView?.metadata?.objects, "labels", "colHeaderFontSize", 11) || 11;
 161         this.colHeaderFontFamily = this.getObjectValue<string>(dataView?.metadata?.objects, "labels", "colHeaderFontFamily", "");
 162         // Reflect checkbox state
 163         const chk = this.toolbar.querySelector('input[type="checkbox"]') as HTMLInputElement | null;
 164         if (chk) chk.checked = this.repeatLabels;
 165         // Load persisted widths and collapsed sets
 166         const widthsJson = this.getObjectValue<string>(dataView?.metadata?.objects, "state", "columnWidths", "");
 167         if (widthsJson) {
 168             try {
 169                 const w = JSON.parse(widthsJson) as { [k: string]: number };
 170                 this.columnWidthPx = new Map(Object.entries(w));
 171             } catch {}
 172         }
 173         const colsJson = this.getObjectValue<string>(dataView?.metadata?.objects, "state", "collapsedCols", "");
 174         if (colsJson) {
 175             try { this.collapsedColKeys = new Set(JSON.parse(colsJson)); this.collapseInitialized = true; } catch {}
 176         }
 177         const rowsJson = this.getObjectValue<string>(dataView?.metadata?.objects, "state", "collapsedRows", "");
 178         if (rowsJson) {
 179             try { this.collapsedRowKeys = new Set(JSON.parse(rowsJson)); this.collapseInitialized = true; } catch {}
 180         }
 181         // If no measure is bound, force all expanded to show full structure
 182         const noMeasures = !(dataView && dataView.matrix && dataView.matrix.valueSources && dataView.matrix.valueSources.length > 0);
 183         if (noMeasures) {
 184             this.collapsedColKeys.clear();
 185             this.collapsedRowKeys.clear();
 186             this.collapseInitialized = true;
 187         }
 188 
 189         // Clear only content (preserve toolbar)
 190         while (this.contentHost.firstChild) this.contentHost.removeChild(this.contentHost.firstChild);
 191 
 192         if (!dataView || !dataView.matrix) {
 193             this.renderPlaceholder("Add Columns hierarchy and a measure");
 194             return;
 195         }
 196 
 197         try {
 198             this.lastMatrix = dataView.matrix;
 199             if (!this.collapseInitialized) {
 200                 const mc = (this.lastMatrix.valueSources && this.lastMatrix.valueSources.length) ? this.lastMatrix.valueSources.length : 0;
 201                 const enableDefault = mc > 0; // don't default-collapse when no measures bound
 202                 this.initializeDefaultCollapsed(this.lastMatrix, enableDefault);
 203                 this.collapseInitialized = true;
 204             }
 205             this.renderMatrix(dataView.matrix);
 206         } catch (e) {
 207             this.renderPlaceholder("Unable to render matrix");
 208             console.error(e);
 209         }
 210     }

 223     private renderMatrix(matrix: DataViewMatrix) {
 224         const columns = matrix.columns;
 225         const rows = matrix.rows;
 226 
 227         if (!columns || !columns.root || !columns.root.children || columns.root.children.length === 0) {
 228             this.renderPlaceholder("No column hierarchy provided");
 229             return;
 230         }
 231 
 232         const table = document.createElement("table");
 233         table.className = "ghm-table";
 234         // Apply a global font if only one family provided
 235         if (this.rowHeaderFontFamily && !this.colHeaderFontFamily) table.style.fontFamily = this.rowHeaderFontFamily;
 236         if (this.colHeaderFontFamily && !this.rowHeaderFontFamily) table.style.fontFamily = this.colHeaderFontFamily;
 237         const colgroup = document.createElement("colgroup");
 238         const thead = document.createElement("thead");
 239         const tbody = document.createElement("tbody");
 240 
 241         // Determine columns to display (compress collapsed groups to a single column)
 242         const colDepth = this.getColumnDepth(columns);
 243         const displayCols = this.computeDisplayColumns(columns.root, colDepth);
 244         // Build column header rows from the display list
 245         const headerRows = this.buildHeaderRowsFromDisplay(displayCols, colDepth);
 246 
 247         // Measures handling: if multiple measures, add an extra header row and expand colspans
 248         const measureCount = (matrix.valueSources && matrix.valueSources.length) ? matrix.valueSources.length : 1;
 249         const measureLabels = (matrix.valueSources || []).map((m, idx) => String((m as any).displayName || (m as any).queryName || `Measure ${idx + 1}`));
 250         if (measureCount > 1) {
 251             for (const row of headerRows) {
 252                 for (const cell of row) cell.span = (cell.span || 1) * measureCount;
 253             }
 254         }
 255 
 256         // Determine number of row header levels (depth)
 257         const rowDepth = this.getRowDepth(rows);
 258         const visibleRowDepth = this.repeatLabels ? rowDepth : this.getVisibleRowDepth(rows, rowDepth);
 259         // Include the thin resizer row as part of the sticky header block
 260         const headerDepth = headerRows.length + (measureCount > 1 ? 1 : 0) + 1;
 261 
 262         headerRows.forEach((rowCells, i) => {
 263             const tr = document.createElement("tr");
 264             if (i === 0 && visibleRowDepth > 0) {
 265                 const corner = document.createElement("th");
 266                 corner.className = "ghm-corner";
 267                 corner.rowSpan = headerDepth;
 268                 if (visibleRowDepth > 1) corner.colSpan = visibleRowDepth;
 269                 tr.appendChild(corner);
 270             }
 271             for (const cell of rowCells) {
 272                 const th = document.createElement("th");
 273                 th.className = "ghm-colheader";
 274                 th.title = cell.label;
 275                 // Toggle indicator and label
 276                 if (cell.togglable) {
 277                     const toggle = document.createElement("span");
 278                     toggle.className = "ghm-toggle";
 279                     toggle.textContent = cell.collapsed ? "+" : "−";
 280                     toggle.addEventListener("click", (ev) => {
 281                         ev.stopPropagation();
 282                         if (cell.collapsed) this.collapsedColKeys.delete(cell.key);
 283                         else this.collapsedColKeys.add(cell.key);
 284                         this.persistState();
 285                         this.refresh();
 286                     });
 287                     th.appendChild(toggle);
 288                 }
 289                 const txt = document.createElement("span");
 290                 txt.textContent = cell.label;
 291                 if (this.colHeaderFontSize) txt.style.fontSize = `${this.colHeaderFontSize}px`;
 292                 if (this.colHeaderFontFamily) txt.style.fontFamily = this.colHeaderFontFamily;
 293                 th.appendChild(txt);
 294                 if (cell.span && cell.span > 1) th.colSpan = cell.span;
 295                 tr.appendChild(th);
 296             }
 297             thead.appendChild(tr);
 298         });
 299 
 300         if (measureCount > 1) {
 301             const tr = document.createElement("tr");
 302             for (const ref of displayCols) {
 303                 for (let m = 0; m < measureCount; m++) {
 304                     const th = document.createElement("th");
 305                     th.className = "ghm-colheader";
 306                     th.textContent = measureLabels[m] ?? `M${m + 1}`;
 307                     tr.appendChild(th);
 308                 }
 309             }
 310             thead.appendChild(tr);
 311         }
 312 
 313         // Resizer row (always present so handles align with columns)
 314         const resizerRow = document.createElement("tr");
 315         resizerRow.className = "ghm-resizers-row";
 316         const resizerRefs = displayCols;
 317         for (const ref of resizerRefs) {
 318             for (let m = 0; m < measureCount; m++) {
 319                 const th = document.createElement("th");
 320                 th.className = "ghm-resizer-cell";
 321                 const key = (ref.kind === "leaf") ? ref.keys.join("||") : ref.key;
 322                 const handle = document.createElement("div");
 323                 handle.className = "ghm-resizer";
 324                 handle.title = "Drag to resize column";
 325                 handle.addEventListener("mousedown", (e) => this.beginResize(e as MouseEvent, key));
 326                 th.appendChild(handle);
 327                 resizerRow.appendChild(th);
 328             }
 329         }
 330         thead.appendChild(resizerRow);
 331 
 332         // Determine column leaf count (for cell generation) and order
 333         const columnLeaves = displayCols;
 334 
 335         // Build colgroup to control widths
 336         this.colElsByKey.clear();
 337         // Prepend only visible row-header columns so grid tightens when collapsed
 338         for (let r = 0; r < visibleRowDepth; r++) {
 339             const col = document.createElement("col");
 340             col.style.width = (r === 0 ? `${this.rowHeaderMinWidth}px` : "120px");
 341             colgroup.appendChild(col);
 342         }
 343         for (const ref of columnLeaves) {
 344             const key = (ref.kind === "leaf") ? ref.keys.join("||") : ref.key;
 345             const width = this.columnWidthPx.get(key) ?? 120;
 346             for (let m = 0; m < measureCount; m++) {
 347                 const col = document.createElement("col");
 348                 col.style.width = `${width}px`;
 349                 colgroup.appendChild(col);
 350                 if (!this.colElsByKey.has(key)) this.colElsByKey.set(key, []);
 351                 this.colElsByKey.get(key)!.push(col);
 352             }
 353         }
 354 
 355         const colLeafCount = columnLeaves.length;
 356 
 357         // Build body rows
 358         if (rows && rows.root && rows.root.children && rows.root.children.length) {
 359             if (this.repeatLabels) {
 360                 const flatRows = this.collectDisplayRowsRepeat(rows.root, rowDepth);
 361                 for (const rowInfo of flatRows) {
 362                     const tr = document.createElement("tr");
 363                     for (let lvl = 0; lvl < rowDepth; lvl++) {
 364                         const th = document.createElement("th");
 365                         th.className = "ghm-rowheader";
 366                         th.textContent = rowInfo.labels[lvl] || "";
 367                         this.applyRowHeaderStyle(th);
 368                         tr.appendChild(th);
 369                     }
 370                     const valuesMap = rowInfo.valuesMap || {} as any;
 371                     const numericKeys = rowInfo.numericKeys || [];
 372                     for (let c = 0; c < colLeafCount; c++) {
 373                         for (let m = 0; m < measureCount; m++) {
 374                             const td = document.createElement("td");
 375                             const v = this.getCellValueForDisplayCol(valuesMap as any, numericKeys, columnLeaves[c], m, measureCount);
 376                             td.textContent = this.formatValue(v);
 377                             tr.appendChild(td);
 378                         }
 379                     }
 380                     tbody.appendChild(tr);
 381                 }
 382             } else {
 383                 const createdRows = this.renderRowGroup(rows.root, 0, "", columnLeaves, measureCount);
 384                 for (const tr of createdRows) tbody.appendChild(tr);
 385             }
 386         } else {
 387             // No row groups – single total row
 388             const tr = document.createElement("tr");
 389             // Data cells
 390             const valuesMap = rows && rows.root ? rows.root.values || {} : {};
 391             const numericKeys = Object.keys(valuesMap as any).map(k => +k).filter(k => !Number.isNaN(k)).sort((a, b) => a - b);
 392             for (let c = 0; c < colLeafCount; c++) {
 393                 for (let m = 0; m < measureCount; m++) {
 394                     const td = document.createElement("td");
 395                     const v = this.getCellValueForDisplayCol(valuesMap as any, numericKeys, columnLeaves[c], m, measureCount);
 396                     td.textContent = this.formatValue(v);
 397                     tr.appendChild(td);
 398                 }
 399             }
 400             tbody.appendChild(tr);
 401         }
 402 
 403         table.appendChild(colgroup);
 404         table.appendChild(thead);
 405         table.appendChild(tbody);
 406         this.contentHost.appendChild(table);
 407         this.applyStickyOffsets(thead);
 408         this.updateDebugOverlay({ table, headerRows, measureCount, columnLeaves, rowDepth, colDepth });
 409     }

 411     private computeDisplayColumns(root: DataViewMatrixNode, depth: number): DisplayCol[] {
 412         type Leaf = { offset: number; labels: string[]; keys: string[]; collapsedAt: number | null; start?: number; end?: number };
 413         const leaves: Leaf[] = [];
 414         const ranges = new Map<string, { start: number; end: number; level: number }>();
 415         let offset = 0;
 416         const walk = (node: DataViewMatrixNode, labels: string[], keys: string[]) => {
 417             const label = this.nodeLabel(node);
 418             const key = [...keys, label].filter(Boolean).join("||");
 419             const newLabels = [...labels, label];
 420             const newKeys = [...keys, label];
 421             if (!node.children || node.children.length === 0) {
 422                 let collapsedAt: number | null = null;
 423                 for (let i = 0; i < newLabels.length; i++) {
 424                     const k = newKeys.slice(0, i + 1).filter(Boolean).join("||");
 425                     if (this.collapsedColKeys.has(k)) { collapsedAt = i; break; }
 426                 }
 427                 while (newLabels.length < depth) newLabels.push("");
 428                 while (newKeys.length < depth) newKeys.push(newKeys[newKeys.length - 1] || "");
 429                 const leaf: Leaf = { offset, labels: newLabels, keys: newKeys, collapsedAt };
 430                 if (collapsedAt !== null) {
 431                     const gkey = newKeys.slice(0, collapsedAt + 1).filter(Boolean).join("||");
 432                     const r = ranges.get(gkey);
 433                     if (!r) ranges.set(gkey, { start: offset, end: offset, level: collapsedAt });
 434                     else r.end = offset;
 435                 }
 436                 leaves.push(leaf);
 437                 offset++;
 438                 return;
 439             }
 440             for (const ch of node.children) walk(ch, newLabels, newKeys);
 441         };
 442         if (root.children) for (const ch of root.children) walk(ch, [], []);
 443 
 444         const result: DisplayCol[] = [];
 445         let i = 0;
 446         while (i < leaves.length) {
 447             const leaf = leaves[i];
 448             if (leaf.collapsedAt !== null) {
 449                 const gkey = leaf.keys.slice(0, leaf.collapsedAt + 1).filter(Boolean).join("||");
 450                 const r = ranges.get(gkey)!;
 451                 result.push({ kind: "collapsed", start: r.start, end: r.end, labels: leaf.labels, keys: leaf.keys, collapsedLevel: r.level, key: gkey });
 452                 i = r.end + 1;
 453             } else {
 454                 result.push({ kind: "leaf", offset: leaf.offset, labels: leaf.labels, keys: leaf.keys });
 455                 i++;
 456             }
 457         }
 458         return result;
 459     }

 461     private isAnyAncestorCollapsed(path: string[]): boolean {
 462         // path is array of labels up to, but not including, the leaf
 463         let accum: string[] = [];
 464         for (const p of path) {
 465             accum.push(p);
 466             const k = accum.join("||");
 467             if (this.collapsedColKeys.has(k)) return true;
 468         }
 469         return false;
 470     }

 472     private getRowDepth(rows?: powerbi.DataViewHierarchy): number {
 473         if (!rows || !rows.root) return 0;
 474         if (rows.levels && rows.levels.length) return rows.levels.length;
 475         const depthFrom = (node: DataViewMatrixNode): number => {
 476             if (!node.children || node.children.length === 0) return 0;
 477             let max = 0;
 478             for (const c of node.children) max = Math.max(max, depthFrom(c));
 479             return 1 + max;
 480         };
 481         if (!rows.root.children || rows.root.children.length === 0) return 0;
 482         return depthFrom(rows.root);
 483     }
 484 
 485     private getColumnDepth(columns?: powerbi.DataViewHierarchy): number {
 486         if (!columns || !columns.root) return 0;
 487         if (columns.levels && columns.levels.length) return columns.levels.length;
 488         const depthFrom = (node: DataViewMatrixNode): number => {
 489             if (!node.children || node.children.length === 0) return 1;
 490             let max = 0;
 491             for (const c of node.children) max = Math.max(max, depthFrom(c));
 492             return 1 + max;
 493         };
 494         if (!columns.root.children || columns.root.children.length === 0) return 1;
 495         return depthFrom(columns.root);
 496     }

 498     private renderRowGroup(node: DataViewMatrixNode, depth: number, parentKey: string, columnLeaves: DisplayCol[], measureCount: number): HTMLTableRowElement[] {
 499         // Skip the artificial root and render its children
 500         if (node.level === undefined && node.children && node.children.length) {
 501             let all: HTMLTableRowElement[] = [];
 502             for (const child of node.children) {
 503                 all = all.concat(this.renderRowGroup(child, 0, parentKey, columnLeaves, measureCount));
 504             }
 505             return all;
 506         }
 507 
 508         if (node.children && node.children.length) {
 509             let rows: HTMLTableRowElement[] = [];
 510             const label = this.nodeLabel(node) || "Total";
 511             const thisKey = [parentKey, label].filter(Boolean).join("||");
 512             const collapsed = this.collapsedRowKeys.has(thisKey);
 513             if (collapsed) {
 514                 // Render a single group row with group aggregates
 515                 const tr = document.createElement("tr");
 516                 const th = document.createElement("th");
 517                 th.className = "ghm-rowheader";
 518                 th.title = label;
 519                 const toggle = document.createElement("span");
 520                 toggle.className = "ghm-toggle";
 521                 toggle.textContent = "+";
 522                 toggle.addEventListener("click", (ev) => {
 523                     ev.stopPropagation();
 524                     this.collapsedRowKeys.delete(thisKey);
 525                     this.persistState();
 526                     this.refresh();
 527                 });
 528                 th.appendChild(toggle);
 529                 const txt = document.createElement("span");
 530                 txt.textContent = label;
 531                 if (this.rowHeaderFontSize) txt.style.fontSize = `${this.rowHeaderFontSize}px`;
 532                 if (this.rowHeaderFontFamily) txt.style.fontFamily = this.rowHeaderFontFamily;
 533                 th.appendChild(txt);
 534                 tr.appendChild(th);
 535                 // fill remaining row header columns (collapsed depths)

 703     private getCollapsedRowGroupValue(node: DataViewMatrixNode, ref: DisplayCol, measureIndex: number, measureCount: number): any {
 704         // Prefer host-provided row subtotal on the collapsed node itself
 705         const tryFromNodeValues = (): any => {
 706             const valuesMap = (node.values || {}) as any;
 707             if (ref.kind === "leaf") {
 708                 const key = ref.offset * measureCount + measureIndex;
 709                 const cell = valuesMap[key];
 710                 if (cell && cell.value != null) return cell.value;
 711                 return undefined;
 712             } else {
 713                 // Collapsed column group: combine the node's per-column-leaf values across the range if present
 714                 let hasAny = false; let total: number | null = null; let first: any = null;
 715                 for (let off = ref.start; off <= ref.end; off++) {
 716                     const key = off * measureCount + measureIndex;
 717                     const cell = valuesMap[key];
 718                     if (cell && cell.value != null) {
 719                         hasAny = true;
 720                         if (typeof cell.value === "number") total = (total ?? 0) + (cell.value as number);
 721                         else if (first === null) first = cell.value;
 722                     }
 723                 }
 724                 if (hasAny) return total !== null ? total : first;
 725                 return undefined;
 726             }
 727         };
 728 
 729         const direct = tryFromNodeValues();
 730         if (direct !== undefined) return direct;
 731         // Otherwise leave empty by design
 732         return "";
 733     }

 735     private aggregateAcrossRowLeaves(node: DataViewMatrixNode, ref: DisplayCol, measureIndex: number, measureCount: number): { sum: number | null; first: any } {
 736         let sum: number | null = null;
 737         let first: any = null;
 738         const visit = (n: DataViewMatrixNode) => {
 739             if (!n.children || n.children.length === 0) {
 740                 const valuesMap = (n.values || {}) as any;

 742                     const key = ref.offset * measureCount + measureIndex;
 743                     const cell = valuesMap[key];
 744                     if (cell && cell.value != null) {
 745                         if (typeof cell.value === "number") sum = (sum ?? 0) + (cell.value as number);
 746                         else if (first === null) first = cell.value;
 747                     }
 748                 } else {
 749                     for (let off = ref.start; off <= ref.end; off++) {
 750                         const key = off * measureCount + measureIndex;
 751                         const cell = valuesMap[key];
 752                         if (cell && cell.value != null) {
 753                             if (typeof cell.value === "number") sum = (sum ?? 0) + (cell.value as number);
 754                             else if (first === null) first = cell.value;
 755                         }
 756                     }
 757                 }

 970     private buildHeaderRowsFromDisplay(displayCols: DisplayCol[], depth: number): Array<Array<{ label: string; span: number; key: string; togglable: boolean; collapsed: boolean }>> {
 971         const rows: Array<Array<{ label: string; span: number; key: string; togglable: boolean; collapsed: boolean }>> = [];
 972         for (let level = 0; level < depth; level++) {
 973             const row: Array<{ label: string; span: number; key: string; togglable: boolean; collapsed: boolean }> = [];
 974             let i = 0;
 975             while (i < displayCols.length) {
 976                 const col = displayCols[i];
 977                 const label = col.kind === "leaf" ? col.labels[level] : (level < col.collapsedLevel ? col.labels[level] : (level === col.collapsedLevel ? col.labels[level] : ""));
 978                 const key = col.keys[level] || "";
 979                 const collapsed = col.kind === "collapsed" && level === col.collapsedLevel ? true : this.collapsedColKeys.has(key) && level < depth - 1;
 980                 const togglable = !!key && level < depth - 1;
 981                 let span = 1;
 982                 let j = i + 1;
 983                 while (j < displayCols.length) {
 984                     const nxt = displayCols[j];
 985                     const nLabel = nxt.kind === "leaf" ? nxt.labels[level] : (level < (nxt.collapsedLevel ?? 0) ? nxt.labels[level] : (level === (nxt.collapsedLevel ?? -1) ? nxt.labels[level] : ""));
 986                     const nKey = nxt.keys[level] || "";
 987                     const nCollapsed = nxt.kind === "collapsed" && level === (nxt.collapsedLevel ?? -1) ? true : this.collapsedColKeys.has(nKey) && level < depth - 1;
 988                     if (nLabel !== label || nKey !== key || nCollapsed !== collapsed) break;
 989                     span++; j++;
 990                 }
 991                 row.push({ label, span, key, togglable, collapsed });
 992                 i = j;
 993             }
 994             rows.push(row);
 995         }
 996         return rows;
 997     }

1041             return;
1042         }
1043         for (const child of node.children) this.collectLeaves(child, out);
1044     }
1045 
1046     private collectRowLeaves(node: DataViewMatrixNode, path: string[], out: Array<{ node: DataViewMatrixNode; label: string }>) {
1047         const label = this.nodeLabel(node);
1048         const nextPath = label ? [...path, label] : [...path];
1049         if (!node.children || node.children.length === 0) {
1050             out.push({ node, label: nextPath.join(" / ") || "Total" });
1051             return;
1052         }
1053         for (const child of node.children) this.collectRowLeaves(child, nextPath, out);
1054     }

1056     private countLeaves(node: DataViewMatrixNode): number {
1057         if (!node.children || node.children.length === 0) return 1;
1058         let n = 0;
1059         for (const child of node.children) n += this.countLeaves(child);
1060         return n;
1061     }

1063     private nodeLabel(node: DataViewMatrixNode): string {
1064         if (node.levelValues && node.levelValues.length) {
1065             // Prefer levelValues for matrix nodes
1066             return String(node.levelValues.map(v => v.value).filter(v => v != null)[0] ?? "");
1067         }
1068         if (node.value != null) return String(node.value);
1069         return "";
1070     }

1072     private formatValue(v: any): string {
1073         if (v == null) return "";
1074         if (typeof v === "number") return v.toLocaleString();
1075         return String(v);
1076     }
